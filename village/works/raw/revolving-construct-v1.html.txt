<script>
(()=>{
"use strict";
const canvas=document.getElementById("world");
const ctx=canvas.getContext("2d");
const msg=document.getElementById("message");
const pause=document.getElementById("pause");
const buttons=[...document.querySelectorAll("[data-mode]")];
const modes={
notice:["Notice","Observe without needing to change anything."],
turn:["Turn","A different angle is another possibility."],
redirect:["Redirect","Energy can become a new direction."],
recover:["Recover","Rest is part of practice, not a failure."],
continue:["Continue","You choose when the next movement begins."]
};
let mode="notice",t=0,stopped=false,last=0;
const reduced=matchMedia("(prefers-reduced-motion: reduce)").matches;
if(reduced){stopped=true;pause.textContent="Resume movement";pause.setAttribute("aria-pressed","true")}
function draw(){
const w=600,c=w/2;
ctx.clearRect(0,0,w,w);
ctx.save();ctx.translate(c,c);
for(let i=1;i<=5;i++){
ctx.beginPath();ctx.arc(0,0,i*48,0,Math.PI*2);
ctx.strokeStyle="rgba(121,220,220,.16)";ctx.lineWidth=1;ctx.stroke();
}
for(let i=0;i<12;i++){
let a=i*Math.PI/6;
ctx.beginPath();ctx.moveTo(Math.cos(a)*35,Math.sin(a)*35);
ctx.lineTo(Math.cos(a)*255,Math.sin(a)*255);
ctx.strokeStyle="rgba(130,190,210,.08)";ctx.stroke();
}
const speed=mode==="recover"?.13:mode==="notice"?.2:mode==="turn"?.65:mode==="redirect"?1.1:.4;
const angle=t*speed;
const radius=mode==="recover"?65:mode==="notice"?115:mode==="redirect"?200:mode==="turn"?170:140;
for(let i=0;i<90;i++){
let a=angle-i*.035;
let r=radius+Math.sin(i*.08+t*.2)*12;
let x=Math.cos(a)*r,y=Math.sin(a)*r;
ctx.fillStyle=`rgba(139,239,214,${(1-i/90)*.65})`;
ctx.beginPath();ctx.arc(x,y,Math.max(.5,5-i*.04),0,Math.PI*2);ctx.fill();
}
let x=Math.cos(angle)*radius,y=Math.sin(angle)*radius;
ctx.shadowBlur=25;ctx.shadowColor="#a8ffe6";
ctx.fillStyle="#c3ffeb";ctx.beginPath();ctx.arc(x,y,11,0,Math.PI*2);ctx.fill();
ctx.shadowBlur=0;
ctx.strokeStyle="#8ddfc8";ctx.lineWidth=2;
ctx.beginPath();ctx.arc(0,0,25,0,Math.PI*2);ctx.stroke();
ctx.fillStyle="#d9fff2";ctx.font="14px system-ui";ctx.textAlign="center";
ctx.fillText("CENTER",0,5);
ctx.restore();
}
function frame(now){
if(!stopped&&last)t+=Math.min((now-last)/1000,.05);
last=now;draw();requestAnimationFrame(frame);
}
buttons.forEach(b=>b.addEventListener("click",()=>{
mode=b.dataset.mode;
msg.textContent=modes[mode][1];
buttons.forEach(x=>x.setAttribute("aria-pressed",String(x===b)));
draw();
}));
pause.addEventListener("click",()=>{
stopped=!stopped;
pause.textContent=stopped?"Resume movement":"Pause movement";
pause.setAttribute("aria-pressed",String(stopped));
draw();
});
buttons[0].setAttribute("aria-pressed","true");
requestAnimationFrame(frame);
})();
</script>
